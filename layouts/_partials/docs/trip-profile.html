{{/* Height profile of all bike rides of the trip as one static SVG: x = ridden kilometres (legs other than bike, and bike legs without an elevation profile, are left out), y = altitude. Every leg is a clickable column linking to the diary post that ends it, in the current language; the tooltip names day, place, distance, ascent and highest point. Works without JavaScript. Data: data/routes///.json (scripts/build_routes.py). Only legs of one trip (docs/trip-key): trip = key of site.Params.trips, default site.Params.defaultTrip. Usage: partial "docs/trip-profile" (dict "pages" site.RegularPages "trip" "singapore") */}} {{- $legs := slice -}} {{- $total := 0.0 -}} {{- $ascent := 0 -}} {{- $minEle := 0.0 -}} {{- $maxEle := 0.0 -}} {{- $maxPage := false -}} {{- $trip := .trip | default site.Params.defaultTrip -}} {{- range sort (where .pages "Section" "docs") "Date" -}} {{- $page := . -}} {{- if ne (partial "docs/trip-key" .) $trip }}{{ continue }}{{ end -}} {{- $src := partial "docs/location-source" . -}} {{- with partial "docs/route-leg" $src -}} {{- $e := .elevation -}} {{- if and (eq .mode "bike") $e -}} {{- $km := float (index (index $e.points (sub (len $e.points) 1)) 0) -}} {{- if gt $km 0.0 -}} {{- $place := $src.Params.location.placeName | default $src.Params.location.name -}} {{- $legs = $legs | append (dict "page" $page "place" $place "e" $e "km" $km "start" $total) -}} {{- $total = add $total $km -}} {{- $ascent = add $ascent $e.ascent -}} {{- $minEle = math.Min $minEle (float $e.min) -}} {{- if gt (float $e.max) $maxEle -}} {{- $maxEle = float $e.max -}} {{- $maxPage = $page -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $legs -}} {{- /* plot area inside the 1000 x 224 viewBox */ -}} {{- $x0 := 44.0 -}} {{- $w := 950.0 -}} {{- $y0 := 10.0 -}} {{- $h := 170.0 -}} {{- $yb := add $y0 $h -}} {{- $ymin := mul (math.Floor (div $minEle 500.0)) 500.0 -}} {{- $ymax := mul (math.Ceil (div $maxEle 1000.0)) 1000.0 -}} {{- $span := sub $ymax $ymin -}} {{- end -}}
