{{/* Map of a set of diary posts: every post with a location becomes a clickable marker linking to the post in the current language, and the leg *ending* at each post (data/routes) is drawn as a line (solid = routed, dashed = straight line, grey = unknown). Renders nothing if none of the pages has a location. Usage: partial "docs/trip-map" (dict "pages" $pages "id" "unique-dom-id" "height" "50vh") */}} {{- $posts := slice -}} {{- $legs := slice -}} {{- $maxPoints := 60 -}} {{- range sort (where .pages "Section" "docs") "Date" -}} {{- $page := . -}} {{- $src := partial "docs/location-source" . -}} {{- with $src.Params.location -}} {{- $posts = $posts | append (dict "lat" .latitude "lon" .longitude "title" $page.LinkTitle "url" $page.RelPermalink "place" .name ) -}} {{- end -}} {{- with partial "docs/route-leg" $src -}} {{- /* thin out long legs, the overview doesn't need every bend */ -}} {{- $step := math.Max 1 (math.Floor (div (len .coordinates) (float $maxPoints))) -}} {{- $last := sub (len .coordinates) 1 -}} {{- $coords := slice -}} {{- range $i, $c := .coordinates -}} {{- if or (eq (mod $i (int $step)) 0) (eq $i $last) -}} {{- /* wrapped in a dict: append would flatten a [lat, lon] slice */ -}} {{- $coords = $coords | append (dict "p" $c) -}} {{- end -}} {{- end -}} {{- $legs = $legs | append (dict "c" $coords "routed" .routed "unknown" (eq .mode "unknown")) -}} {{- end -}} {{- end -}} {{ if $posts }} {{ $id := .id }} {{ $height := .height | default "70vh" }}
{{ end }}
